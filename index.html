/** * Evaluates field data to return decision board flags for a specific runner. * @param {Object} runner - Target runner data. * @param {Array} field - Complete list of runner objects in the race. * @returns {Array} Array of badge objects with type, label, color, and reason. */ function getDecisionBoardFlags(runner, field) { const flags = []; // --- 1. OVERLAY ALERT --- // Compares fair odds derived from the model against Morning Line (ML) odds if (runner.mlOddsDecimal && runner.fairOddsDecimal) { const valueRatio = runner.mlOddsDecimal / runner.fairOddsDecimal; if (valueRatio >= 1.35) { flags.push({ type: 'OVERLAY', label: '🎯 OVERLAY', color: '#2e7d32', // Green detail: `${valueRatio.toFixed(1)}x Value (ML ${runner.mlOdds} vs Fair ${runner.fairOdds})` }); } } // --- 2. LONE SPEED FLAG --- // Evaluates Early Pace (E-Pace) differential relative to second-highest in field const sortedByPace = [...field].sort((a, b) => b.ePaceRating - a.ePaceRating); if (sortedByPace[0] && sortedByPace[0].id === runner.id) { const secondHighestPace = sortedByPace[1] ? sortedByPace[1].ePaceRating : 0; const paceGap = runner.ePaceRating - secondHighestPace; if (paceGap >= 7) { flags.push({ type: 'LONE_SPEED', label: '⚡ LONE SPEED', color: '#e65100', // Orange detail: `+${paceGap} E-Pace differential over field` }); } } // --- 3. VULNERABLE FAVORITE WARNING --- // Identifies ML Favorite with poor recent indicators const isMLFavorite = field.every(r => runner.mlOddsDecimal <= r.mlOddsDecimal); if (isMLFavorite) { const longLayoff = runner.daysOff > 150; const speedDeclining = runner.lastSpeedFig < (runner.avgSpeedFig - 8); const sharpClassRise = runner.classChangeStep < -2; if (longLayoff || speedDeclining || sharpClassRise) { const reason = longLayoff ? `${runner.daysOff}d Layoff` : speedDeclining ? 'Form Regressing' : 'Major Class Step Up'; flags.push({ type: 'VULNERABLE_FAV', label: '⚠️ VULNERABLE FAV', color: '#c62828', // Red detail: `Low-value ML Favorite (${reason})` }); } } return flags; }
